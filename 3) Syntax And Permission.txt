Calculate Percentage of 5 Subject Marks Taken as A User Input

set serveroutput on;
DECLARE
	dbms number(3);
	ai number(3);
	python number(3);
	maths number(3);
	english number(3); 
	marks number(3);
	percentage real(10);
BEGIN
	dbms:=&DBMS;
	ai:=&AI;
	python:=&PYTHON;
	maths:=&MATHS;
	english:=&ENGLISH;
	marks:=(dbms+ai+python+maths+english);
	percentage:=marks/5;
	dbms_output.put_line(marks);
	dbms_output.put_line(percentage);
END;
